## 状态码速查

| status | status_msg | 含义 |
|--------|-----------|------|
| 3 | 已报 | 订单已挂入订单簿，等待成交 |
| 0 | 全部成交 | 订单全部成交，从队列移除 |
| 1 | 部分成交 | 部分成交，剩余继续排队 |
| 5 | 已撤单 | 撤单成功 |

## 编写策略

```python
from hft_backtest import BacktestEngine, CtaTemplate

class MyStrategy(CtaTemplate):
    def on_tick(self, tick):
        if tick.symbol != self.vt_symbol:
            return
        # 你的逻辑
        if tick.last_price < 100:
            self.buy(price=tick.last_price, volume=1)

    def on_order(self, order):
        st = {'0':'成交','1':'部分成交','3':'排队','5':'撤单'}.get(order.status, order.status)
        self.write_log(f"[回报] {order.order_sys_id} {order.direction}{order.offset} "
                       f"@{order.price} 成交{order.volume_traded}/{order.volume_total} [{st}]")

    def on_trade(self, trade):
        self.write_log(f"[成交] {trade.order_sys_id} {trade.direction}{trade.offset} "
                       f"@{trade.price} vol={trade.volume}")

engine = BacktestEngine()
engine.set_parameters(
    data_mode="tick",
    data_file="data/market_data_20260523_night",
    symbols_file="conf/symbols.txt",
    initial_balance=1_000_000,
    output_dir="results/my_test",
)
engine.add_strategy(MyStrategy, {"symbol": "au2606"})
result = engine.run_backtesting()
```
